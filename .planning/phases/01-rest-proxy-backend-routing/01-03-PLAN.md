```python
class DiskCache:
    """LRU-pruned, size-capped, atomic-write file cache.
    Silently disables itself if the cache dir is not writable (Splunk Cloud)."""

    def __init__(self, cache_dir: str, max_bytes: int = 500 * 1024 * 1024):
        """Attempts os.makedirs(cache_dir). On PermissionError or OSError,
        sets self.enabled = False and logs INFO once. Otherwise enabled=True."""

    def get(self, key: str) -> tuple | None:
        """Returns (bytes, content_type, cache_control) or None on miss.
        Updates file mtime on hit (touch) so LRU prune sees it as recently used.
        Returns None on any IOError (degrades gracefully)."""

    def set(self, key: str, value: tuple) -> None:
        """value is (bytes, content_type, cache_control). Serializes:
        4-byte BE length + content_type utf8 + 4-byte BE length +
        cache_control utf8 + raw bytes. Atomic via tmp+os.replace.
        Triggers _prune if total cache size exceeds max_bytes."""

    def _prune(self) -> None:
        """Scans cache dir, sorts by mtime ascending, deletes oldest until
        total size <= max_bytes. Holds self._lock throughout."""

    def _path_for(self, key: str) -> str:
        """Returns cache_dir/<first-2-hex>/<key>.tile. Does NOT create dirs."""

    def _assert_within_cache_dir(self, path: str) -> None:
        """Raises ValueError if realpath(path) is not under realpath(cache_dir).
        Called before every disk write (S-10 / T1-05)."""
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| In-process memory → disk | Cache writes cross process→FS; path-traversal risk |
| Filesystem → in-process memory | Cache reads trust that on-disk bytes are what we wrote (no tamper-proofing) |
| Multiple threads → single disk index | Splunk `scripttype = persist` may call `handle_GET` concurrently |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T3-01-PathTraversal | Tampering | `DiskCache._path_for` / writes | mitigate | Keys are `hashlib.sha256(url).hexdigest()` — 64 hex chars, no traversal possible from URL input. Defense-in-depth: `_assert_within_cache_dir` checks `os.path.realpath(file).startswith(os.path.realpath(cache_dir))` before every open() (S-10). Inherits from T1-05 |
| T3-02-TornWrite | Integrity | `DiskCache.set` | mitigate | Write to `<final>.tmp` then `os.replace(tmp, final)` (atomic on POSIX + Windows Python 3.3+). Addresses audit A-08 |
| T3-03-ConcurrencyRace | Integrity | `DiskCache` index + prune | mitigate | Single `threading.Lock` around `get`/`set`/`_prune`. Addresses audit A-07. The qwen plan's claim of no-lock-needed is wrong for `scripttype = persist` (RESEARCH Pattern 5 note) |
| T3-04-DiskExhaustion | DoS | unbounded cache growth | mitigate | `_prune` enforces `disk_cache_max_mb` cap on every `set`. LRU eviction by file mtime. Default 500MB (DS-CL-02) |
| T3-05-CloudFilesystem | Availability | Splunk Cloud write restrictions | mitigate | On `os.makedirs` `PermissionError` or `OSError(errno.EROFS)`, set `self.enabled = False`, log INFO once, return None from all methods. Handler stays functional with memory-only LRU. Addresses audit A-06 |
| T3-06-SerializationFormat | Integrity | cache file format | mitigate | Custom length-prefixed binary format: 4-byte BE length + UTF-8 content_type + 4-byte BE length + UTF-8 cache_control + raw bytes. Malformed files → `get` returns None (corrupt file treated as miss). No `pickle` (RCE risk if attacker writes to cache dir) |
| T3-07-SymlinkEscape | Tampering | cache dir contents | mitigate | Use `os.open(path, O_WRONLY|O_CREAT|O_NOFOLLOW)` where supported; or at minimum, `_assert_within_cache_dir` uses `realpath` which resolves symlinks before the prefix check |
| T3-08-MtimeUpdateRace | Reliability | LRU ordering accuracy | accept | `os.utime()` touch on read is best-effort; two concurrent reads racing on utime is benign (both set near-current time). Documented, no lock needed for utime |
</threat_model>

    ```python
    class DiskCache(object):
        """Size-capped, atomically-written, path-confined LRU disk cache.

        File format: [magic 4B][ct_len 4B BE][content_type utf8]
                     [cc_len 4B BE][cache_control utf8][data ...]
        Cache path : <cache_dir>/<key[:2]>/<key>.tile
        """

        def __init__(self, cache_dir, max_bytes=DEFAULT_DISK_CACHE_MAX_MB * 1024 * 1024):
            self.cache_dir = cache_dir
            self.max_bytes = int(max_bytes)
            self._lock = threading.Lock()
            self.enabled = False
            try:
                os.makedirs(cache_dir, exist_ok=True)
                # Probe: write + delete a tiny marker file to confirm writability
                probe = os.path.join(cache_dir, ".writeprobe")
                with open(probe, "wb") as f:
                    f.write(b"ok")
                os.remove(probe)
                self.enabled = True
            except (PermissionError, OSError) as e:
                # Splunk Cloud read-only FS path (audit A-06)
                if isinstance(e, OSError) and getattr(e, "errno", None) == errno.EROFS:
                    logger.info("disk_cache_disabled_readonly_fs dir=%s", cache_dir)
                else:
                    logger.info("disk_cache_disabled reason=%s dir=%s",
                                type(e).__name__, cache_dir)
                self.enabled = False

        # ---- path helpers ----

        def _path_for(self, key):
            return os.path.join(self.cache_dir, key[:2], key + ".tile")

        def _assert_within_cache_dir(self, path):
            real_path = os.path.realpath(path)
            real_base = os.path.realpath(self.cache_dir)
            if not (real_path == real_base or
                    real_path.startswith(real_base + os.sep)):
                raise ValueError("path_escape_detected")

        # ---- public API ----

        def get(self, key):
            if not self.enabled:
                return None
            path = self._path_for(key)
            try:
                self._assert_within_cache_dir(path)
                with open(path, "rb") as f:
                    header = f.read(4)
                    if header != _DISK_CACHE_FORMAT_VERSION:
                        return None
                    ct_len_bytes = f.read(4)
                    if len(ct_len_bytes) < 4:
                        return None
                    ct_len = struct.unpack(">I", ct_len_bytes)[0]
                    if ct_len > 256:
                        return None
                    content_type = f.read(ct_len).decode("utf-8", "replace")
                    cc_len_bytes = f.read(4)
                    if len(cc_len_bytes) < 4:
                        return None
                    cc_len = struct.unpack(">I", cc_len_bytes)[0]
                    if cc_len > 256:
                        return None
                    cache_control = f.read(cc_len).decode("utf-8", "replace")
                    data = f.read(MAX_TILE_BYTES + 1)
                    if len(data) > MAX_TILE_BYTES:
                        return None
                # LRU touch — best-effort; ignore errors
                try:
                    now = time.time()
                    os.utime(path, (now, now))
                except OSError:
                    pass
                return (data, content_type, cache_control)
            except (FileNotFoundError, IsADirectoryError, ValueError):
                return None
            except OSError:
                return None

        def set(self, key, value):
            if not self.enabled:
                return
            data, content_type, cache_control = value
            if not isinstance(data, bytes):
                raise TypeError("data must be bytes")
            ct_bytes = (content_type or "application/octet-stream").encode("utf-8")[:256]
            cc_bytes = (cache_control or DEFAULT_CACHE_CONTROL).encode("utf-8")[:256]
            path = self._path_for(key)
            self._assert_within_cache_dir(path)
            with self._lock:
                try:
                    subdir = os.path.dirname(path)
                    os.makedirs(subdir, exist_ok=True)
                    # Write to a unique tmp file then atomic replace
                    tmp_fd, tmp_path = tempfile.mkstemp(
                        prefix=".tmp_", suffix=".tile", dir=subdir)
                    try:
                        with os.fdopen(tmp_fd, "wb") as f:
                            f.write(_DISK_CACHE_FORMAT_VERSION)
                            f.write(struct.pack(">I", len(ct_bytes)))
                            f.write(ct_bytes)
                            f.write(struct.pack(">I", len(cc_bytes)))
                            f.write(cc_bytes)
                            f.write(data)
                            f.flush()
                            try:
                                os.fsync(f.fileno())
                            except OSError:
                                pass
                        os.replace(tmp_path, path)   # atomic
                    except Exception:
                        try: os.remove(tmp_path)
                        except OSError: pass
                        raise
                    self._prune_locked()
                except (PermissionError, OSError) as e:
                    logger.warning("disk_cache_set_failed key=%s err=%s",
                                   key[:8], type(e).__name__)

        def _prune_locked(self):
            """Caller must hold self._lock. Removes oldest-mtime files until
            total size <= self.max_bytes."""
            entries = []
            total = 0
            for root, _dirs, files in os.walk(self.cache_dir):
                for name in files:
                    if not name.endswith(".tile"):
                        continue
                    p = os.path.join(root, name)
                    try:
                        st = os.stat(p)
                    except OSError:
                        continue
                    entries.append((st.st_mtime, st.st_size, p))
                    total += st.st_size
            if total <= self.max_bytes:
                return
            entries.sort()   # ascending mtime = oldest first
            for mtime, size, p in entries:
                if total <= self.max_bytes:
                    break
                try:
                    os.remove(p)
                    total -= size
                except OSError:
                    continue
    ```
