      test('leaves {z} {x} {y} untouched for server resolution', () => {
        const t = 'https://tile.osm.org/{z}/{x}/{y}.png';
        expect(h.normalizeTileTemplate(t, {})).toBe(t);
      });

          gibsTileMatrixSet: '500m',
          gibsFormat: 'jpg'

        expect(out).toContain('BlueMarble_ShadedRelief_Bathymetry');
        expect(out).toContain('2023-01-01');
        expect(out).toContain('500m');

          __proto__: 'pwn',

        expect(h.normalizeTileTemplate(null, {})).toBe(null);

      test('produces the route shape /maps_plus/tile/proxy?url=<enc>&z=&x=&y=', () => {
        const u = h.buildTileProxyUrl('/services', 'https://tile.osm.org/{z}/{x}/{y}.png', { z: 3, x: 2, y: 1 });

      });
      test('URL-encodes the template exactly once (T2-06 no double-encode)', () => {

        expect(u).toContain('url=https%3A%2F%2Fx.com%2F%3Fa%3D1%26b%3D2');
        expect(u).not.toContain('%25');
      });
      test('appends optional s and r query params when provided (D-10)', () => {
        const u = h.buildTileProxyUrl('/services', 'https://{s}.tile/{z}/{x}/{y}.png', { z: 1, x: 1, y: 1 }, { s: 'a', r: '2' });

        expect(u).toContain('&r=2');

        const u = h.buildTileProxyUrl('/services', 'https://tile/{z}/{x}/{y}.png', { z: 0, x: 0, y: 0 });

        expect(u).not.toContain('&r=');
      });
      test('omits s/r when empty string passed (empty is not sent)', () => {
        const u = h.buildTileProxyUrl('/services', 'https://tile/{z}/{x}/{y}.png', { z: 0, x: 0, y: 0 }, { s: '', r: '' });

        expect(u).not.toContain('&r=');
      });
      test('trims trailing slashes from restRoot', () => {
        const u = h.buildTileProxyUrl('/services/', 'https://tile/{z}/{x}/{y}.png', { z: 0, x: 0, y: 0 });
